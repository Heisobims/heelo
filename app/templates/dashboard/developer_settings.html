1. Authenticate API Requests

Pass your key via the X-API-Key request header. All responses are formatted as standard JSON. 

curl -X GET "https://api.heelo.io/api/public/v1/events" \
  -H "X-API-Key: hlo_live_YOUR_SECRET_KEY" \
  -H "Accept: application/json"
import requests

API_KEY = "hlo_live_YOUR_SECRET_KEY"
url = "https://api.heelo.io/api/public/v1/events"

headers = {
    "X-API-Key": API_KEY,
    "Accept": "application/json",
}

response = requests.get(url, headers=headers)
events = response.json()
print("Events:", events)
const API_KEY = "hlo_live_YOUR_SECRET_KEY";

async function fetchEvents() {
  const response = await fetch("https://api.heelo.io/api/public/v1/events", {
    headers: {
      "X-API-Key": API_KEY,
      "Accept": "application/json",
    },
  });
  const data = await response.json();
  console.log("Events:", data);
}

fetchEvents();

2. Verify Incoming Webhook Signatures

Heelo signs payloads using HMAC-SHA256 with the subscription secret. Header format: X-Heelo-Signature: t=1697750000,v1=hash. 

# Verify incoming X-Heelo-Signature: t=<timestamp>,v1=<signature>
# 1. Split header into timestamp `t` and signature `v1`
# 2. Compute HMAC-SHA256 with secret on `<timestamp>.<raw_json_body>`
# 3. Compare computed digest against `v1` using constant-time comparison.
import hmac
import hashlib

WEBHOOK_SECRET = "whsec_YOUR_SIGNING_SECRET"

def verify_heelo_webhook(raw_payload_bytes: bytes, sig_header: str) -> bool:
    try:
        parts = dict(pair.split("=", 1) for pair in sig_header.split(","))
        timestamp, signature = parts["t"], parts["v1"]
    except (ValueError, KeyError):
        return False

    # Signature is calculated on `timestamp.body`
    signed_payload = f"{timestamp}.".encode() + raw_payload_bytes
    expected_sig = hmac.new(
        WEBHOOK_SECRET.encode(),
        signed_payload,
        hashlib.sha256
    ).hexdigest()

    return hmac.compare_digest(expected_sig, signature)
const crypto = require('crypto');

const WEBHOOK_SECRET = "whsec_YOUR_SIGNING_SECRET";

function verifyHeeloWebhook(rawBodyBuffer, signatureHeader) {
  try {
    const parts = Object.fromEntries(
      signatureHeader.split(',').map(kv => kv.split('='))
    );
    const { t: timestamp, v1: signature } = parts;

    const hmac = crypto.createHmac('sha256', WEBHOOK_SECRET);
    hmac.update(`${timestamp}.`);
    hmac.update(rawBodyBuffer);
    const expectedSig = hmac.digest('hex');

    return crypto.timingSafeEqual(
      Buffer.from(signature, 'hex'),
      Buffer.from(expectedSig, 'hex')
    );
  } catch (err) {
    return false;
  }
}
